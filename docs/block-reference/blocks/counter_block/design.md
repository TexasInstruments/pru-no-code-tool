    ldi32 TEMP_REG1, 0x11
    sbbo  &TEMP_REG1, TEMP_REG4, 0, 4

    add   RES, RES, 1
    .endif
    qba   ct_ih?
ct_ih?:
    lbbo  &TEMP_REG2, TEMP_REG4, 0x10, 4
    qbge  ct_ge?, TEMP_REG1, TEMP_REG2
    qbbs  ct_ih?, R31, INPUT_PIN

    .endif
