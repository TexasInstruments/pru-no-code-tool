## Generated macro (`.xdt` equivalent)

```asm
m_mac_hw .macro RES, OP_A, OP_B, ACC_VAL
    ; --- MAC Hardware ---
    ; Configure MAC for multiply-only mode
    ldi32 R25, 0
    xout  0, &R25, 1
    ; Load operands to MAC input registers
    mov   R28, OP_A
    mov   R29, OP_B
    nop
    ; Read result (lower 32-bits from R26)
    xin   0, &TEMP_REG3, 4
    ; Accumulate
    add   RES, TEMP_REG3, ACC_VAL
    .endm
```
